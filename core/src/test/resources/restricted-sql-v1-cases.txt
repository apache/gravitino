in	row	(x STRING) RETURNS BOOLEAN -> x IN ('US', 'CA', NULL)
not_in	row	(x STRING) RETURNS BOOLEAN -> x NOT IN ('US', NULL)
between	row	(x STRING) RETURNS BOOLEAN -> x BETWEEN 'A' AND 'Z'
not_between	row	(x STRING) RETURNS BOOLEAN -> x NOT BETWEEN 'A' AND 'Z'

time	row	(x TIME) RETURNS BOOLEAN -> x = TIME '12:30:00.123456'
timestamp	row	(x TIMESTAMP) RETURNS BOOLEAN -> x = TIMESTAMP '2026-01-01 12:30:00'

long_max	row	(x LONG) RETURNS BOOLEAN -> x = 9223372036854775807
decimal	row	(x DECIMAL(4,2)) RETURNS BOOLEAN -> x = 99.99
mask_alphanum	mask	(value STRING) RETURNS STRING -> mask_alphanum(value)
show_first_4	mask	(value STRING) RETURNS STRING -> show_first_4(value)

sha256_global	mask	(value STRING) RETURNS STRING -> sha256_global(value)
